   262→   262→                            freedBytes += file.length()

   340→   340→    fun updateCleanupSchedule(context: Context) {
   341→   341→        try {
   342→   342→            val workManager = androidx.work.WorkManager.getInstance(context)
   343→   343→            val enabled = _cleanupEnabled.value
   344→   344→            val frequency = _cleanupFrequency.value
   345→   345→
   346→   346→            workManager.cancelAllWorkByTag("storage_cleanup")
   347→   347→
   348→   348→            if (enabled) {
   349→   349→                val repeatInterval = when (frequency) {
   350→   350→                    "Daily" -> 1L
   351→   351→                    "Weekly" -> 7L
   352→   352→                    "Monthly" -> 30L
   353→   353→                    else -> 7L
   354→   354→                }
   355→   355→                val constraints = androidx.work.Constraints.Builder()
   356→   356→                    .setRequiredNetworkType(androidx.work.NetworkType.CONNECTED)
   357→   357→                    .setRequiresBatteryNotLow(true)
   358→   358→                    .build()
   359→   359→
   360→   360→                val cleanupRequest = androidx.work.PeriodicWorkRequestBuilder<com.example.worker.StorageCleanupWorker>(
   361→   361→                    repeatInterval, java.util.concurrent.TimeUnit.DAYS
   362→   362→                )
   363→   363→                    .setConstraints(constraints)
   364→   364→                    .addTag("storage_cleanup")
   365→   365→                    .build()
   366→   366→
   367→   367→                workManager.enqueueUniquePeriodicWork(
   368→   368→                    "storage_cleanup_work",
   369→   369→                    androidx.work.ExistingPeriodicWorkPolicy.REPLACE,
   370→   370→                    cleanupRequest
   371→   371→                )
   372→   372→                android.util.Log.d("SettingsViewModel", "Scheduled storage cleanup: $frequency")
   373→   373→            } else {
   374→   374→                android.util.Log.d("SettingsViewModel", "Automatic storage cleanup disabled.")
   375→   375→            }
   376→   376→        } catch (e: Throwable) {
   377→   377→            android.util.Log.e("SettingsViewModel", "Failed to schedule storage cleanup", e)
   378→   378→        }
   379→   379→    }
