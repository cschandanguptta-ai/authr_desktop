   188→    fun autoFetchModels(targetProvider: String? = null) {
   189→        val prov = targetProvider ?: _selectedProvider.value
   190→        val key = when (prov) {
   191→            "Gemini" -> _geminiApiKey.value.ifEmpty { com.example.BuildConfig.GEMINI_API_KEY }
   192→            "Mistral" -> _mistralApiKey.value
   193→            "OpenRouter" -> _openRouterApiKey.value
   194→            "Groq" -> _groqApiKey.value
   195→            "NVIDIA" -> _nvidiaApiKey.value
   196→            "Custom OpenAI" -> _customOpenAiApiKey.value
   197→            "Ollama" -> _ollamaApiKey.value.ifEmpty { com.example.BuildConfig.OLLAMA_API_KEY }
   198→            else -> ""
   199→        }
   200→        val baseUrl = when (prov) {
   201→            "Ollama" -> _ollamaBaseUrl.value
   202→            "Custom OpenAI" -> _customOpenAiBaseUrl.value
   203→            else -> ""
   204→        }
   205→
   206→        viewModelScope.launch {
   207→            _isFetchingModels.value = true
   208→            _modelFetchStatus.value = "Fetching live models for $prov..."
   209→            val result = com.example.data.api.ModelDiscoveryService.fetchModelsForProvider(prov, key, baseUrl)
   210→            val models = result.getOrDefault(com.example.data.api.ModelDiscoveryService.getFallbackModelsForProvider(prov))
   211→            val currentMap = _fetchedModels.value.toMutableMap()
   212→            currentMap[prov] = models
   213→            _fetchedModels.value = currentMap
   214→            _isFetchingModels.value = false
   215→            _modelFetchStatus.value = "Discovered ${models.size} models for $prov"
   216→            sharedPrefs.edit().putString("cached_models_$prov", models.joinToString(",")).apply()
   217→        }
   218→    }

   246→    fun pruneAudioCache(context: Context) {
   247→        viewModelScope.launch(kotlinx.coroutines.Dispatchers.IO) {
   248→            try {
   249→                var freedBytes = 0L
   250→                val cacheDir = context.cacheDir
   251→                cacheDir.walkTopDown().forEach { file ->
   252→                    if (file.isFile && (file.extension.equals("wav", ignoreCase = true) || file.name.startsWith("tts_") || file.name.startsWith("kittentts_"))) {
   253→                        freedBytes += file.length()
   254→                        file.delete()
   255→                    }
   256→                }
   257→                val downloadsDir = android.os.Environment.getExternalStoragePublicDirectory(android.os.Environment.DIRECTORY_DOWNLOADS)
   258→                val weaverDir = java.io.File(downloadsDir, "Weaver_Studio/AudioCache")
   259→                if (weaverDir.exists()) {
   260→                    weaverDir.walkTopDown().forEach { file ->
   261→                        if (file.isFile) {
   262→                            freedBytes += file.length()
   263→                            file.delete()
   264→                        }
   265→                    }
   266→                }
   267→                _audioCacheSize.value = 0L
   268→                kotlinx.coroutines.withContext(kotlinx.coroutines.Dispatchers.Main) {
   269→                    android.widget.Toast.makeText(context, "🧹 Voice Cache Pruned: Freed ${freedBytes / 1024} KB of storage", android.widget.Toast.LENGTH_SHORT).show()
   270→                }
   271→            } catch (e: Exception) {
   272→                android.util.Log.e("SettingsViewModel", "Failed to prune audio cache", e)
   273→            }
   274→        }
   275→    }

   340→    fun updateCleanupSchedule(context: Context) {
   341→        try {
   342→            val workManager = androidx.work.WorkManager.getInstance(context)
   343→            val enabled = _cleanupEnabled.value
   344→            val frequency = _cleanupFrequency.value
   345→
   346→            workManager.cancelAllWorkByTag("storage_cleanup")
   347→
   348→            if (enabled) {
   349→                val repeatInterval = when (frequency) {
   350→                    "Daily" -> 1L
   351→                    "Weekly" -> 7L
   352→                    "Monthly" -> 30L
   353→                    else -> 7L
   354→                }
   355→                val constraints = androidx.work.Constraints.Builder()
   356→                    .setRequiredNetworkType(androidx.work.NetworkType.CONNECTED)
   357→                    .setRequiresBatteryNotLow(true)
   358→                    .build()
   359→
   360→                val cleanupRequest = androidx.work.PeriodicWorkRequestBuilder<com.example.worker.StorageCleanupWorker>(
   361→                    repeatInterval, java.util.concurrent.TimeUnit.DAYS
   362→                )
   363→                    .setConstraints(constraints)
   364→                    .addTag("storage_cleanup")
   365→                    .build()
   366→
   367→                workManager.enqueueUniquePeriodicWork(
   368→                    "storage_cleanup_work",
   369→                    androidx.work.ExistingPeriodicWorkPolicy.REPLACE,
   370→                    cleanupRequest
   371→                )
   372→                android.util.Log.d("SettingsViewModel", "Scheduled storage cleanup: $frequency")
   373→            } else {
   374→                android.util.Log.d("SettingsViewModel", "Automatic storage cleanup disabled.")
   375→            }
   376→        } catch (e: Throwable) {
   377→            android.util.Log.e("SettingsViewModel", "Failed to schedule storage cleanup", e)
   378→        }
   379→    }
