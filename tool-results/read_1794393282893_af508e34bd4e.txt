    39→    fun escapeXml(text: String): String {
    40→        if (text.isEmpty()) return ""
    41→        return text
    42→            .replace("&", "&amp;")
    43→            .replace("<", "&lt;")
    44→            .replace(">", "&gt;")
    45→            .replace("\"", "&quot;")
    46→            .replace("'", "&apos;")
    47→    }

    67→    /**
    68→     * Export complete book package as a compressed .zip archive containing TXT, Markdown, Story Bible, and Outline.
    69→     */
    70→    fun exportBookToZipPackage(context: Context, bookTitle: String, manuscript: String, bible: String, outline: String): String? {
    71→        return try {
    72→            val downloads = Environment.getExternalStoragePublicDirectory(Environment.DIRECTORY_DOWNLOADS)
    73→            val studioDir = File(downloads, "Weaver_Studio")
    74→            studioDir.mkdirs()
    75→            val safeTitle = bookTitle.replace(Regex("[^a-zA-Z0-9_\\-]"), "_")
    76→            val zipFile = File(studioDir, "${safeTitle}_Package.zip")
    77→            ZipOutputStream(zipFile.outputStream().buffered()).use { zip ->
    78→                zip.putNextEntry(ZipEntry("Manuscript.txt"))
    79→                zip.write(manuscript.toByteArray())
    80→                zip.closeEntry()
    81→
    82→                zip.putNextEntry(ZipEntry("Manuscript.md"))
    83→                zip.write("# $bookTitle\n\n$manuscript".toByteArray())
    84→                zip.closeEntry()
    85→
    86→                zip.putNextEntry(ZipEntry("StoryBible.txt"))
    87→                zip.write(bible.toByteArray())
    88→                zip.closeEntry()
    89→
    90→                zip.putNextEntry(ZipEntry("Outline.txt"))
    91→                zip.write(outline.toByteArray())
    92→                zip.closeEntry()
    93→            }
    94→            zipFile.absolutePath
    95→        } catch (e: Exception) {
    96→            Log.e("EpubManager", "Failed to export ZIP package", e)
    97→            null
    98→        }
    99→    }

   101→    /**
   102→     * Validate the saved cover image's resolution and aspect ratio
   103→     */
   104→    fun validateCoverImage(context: Context, storyId: Int): ImageValidationResult {
   105→        val coverFile = File(context.filesDir, "story_${storyId}_cover.jpg")
   106→        if (!coverFile.exists()) {
   107→            return ImageValidationResult(
   108→                isValid = false,
   109→                width = 0,
   110→                height = 0,
   111→                aspectRatio = 0f,
   112→                warningMessage = "No cover image file uploaded. The eBook will use a dynamically generated typography cover."
   113→            )
   114→        }
   115→        return try {
   116→            val options = BitmapFactory.Options().apply { inJustDecodeBounds = true }
   117→            BitmapFactory.decodeFile(coverFile.absolutePath, options)
   118→            val width = options.outWidth
   119→            val height = options.outHeight
   120→            if (width <= 0 || height <= 0) {
   121→                ImageValidationResult(
   122→                    isValid = false,
   123→                    width = 0,
   124→                    height = 0,
   125→                    aspectRatio = 0f,
   126→                    warningMessage = "Invalid image format or corrupt file."
   127→                )
   128→            } else {
   129→                val ratio = height.toFloat() / width
   130→                var warning: String? = null
   131→                if (width < 1400 || height < 2100) {
   132→                    warning = "Low resolution: Recommended cover resolution is at least 1400x2100 px for professional platforms (current: ${width}x${height} px)."
   133→                } else if (ratio < 1.4f || ratio > 1.7f) {
   134→                    warning = "Suboptimal aspect ratio: Recommended aspect ratio is 1:1.5 to 1:1.6 (current: 1:${String.format(Locale.US, "%.2f", ratio)}). Cover may scale awkwardly on some eReaders."
   135→                }
   136→                ImageValidationResult(
   137→                    isValid = true,
   138→                    width = width,
   139→                    height = height,
   140→                    aspectRatio = ratio,
   141→                    warningMessage = warning
   142→                )
   143→            }
   144→        } catch (e: Exception) {
   145→            ImageValidationResult(
   146→                isValid = false,
   147→                width = 0,
   148→                height = 0,
   149→                aspectRatio = 0f,
   150→                warningMessage = "Error validating cover image: ${e.message}"
   151→            )
   152→        }
   153→    }

   158→    fun generateCoverPrompt(story: Story): String {

   163→        val extras = story.extras.ifEmpty { "" }
   164→        val mood = when {
   165→            genres.contains("Thriller", true) || genres.contains("Horror", true) -> "dark, mysterious, suspenseful, tense"
   166→            genres.contains("Fantasy", true) || genres.contains("Sci-Fi", true) -> "epic, mystical, wonderous, celestial"
   167→            genres.contains("Romance", true) -> "emotional, romantic, intimate, soft, warm"

   170→        val artStyle = when {
   171→            genres.contains("Fantasy", true) -> "Dramatic digital fantasy illustration, high-fantasy style, highly detailed"
   172→            genres.contains("Sci-Fi", true) -> "Futuristic cinematic concept art, sci-fi cyberpunk aesthetic, sleek rendering"
   173→            genres.contains("Thriller", true) -> "Chiaroscuro film noir, dramatic shadows, mysterious realism"
   174→            genres.contains("Romance", true) -> "Soft focus impressionistic digital painting, warm pastel color palette"
   175→            else -> "Cinematic oil painting style, masterpiece digital art, volumetric lighting"
   176→        }
   177→        
   178→        return """
   179→            EBook Cover Art Concept for "$title" (Genre: $genres)
   180→            --------------------------------------------------
   181→            Artistic Style: $artStyle.
   182→            Subject & Character: A highly detailed depiction of $hero, matching the central premise: "$hook".
   183→            Atmosphere & Mood: $mood, evocative, captivating.
   184→            Color Palette: Masterful lighting with contrasting tones, color-graded to match the thematic depth of the story.
   185→            Composition & Camera Angle: Centered single-focus composition with generous negative space at the top and bottom for title typography placement. High contrast between subject and background.
   186→            Visual Symbolism: Incorporating elements representing: $extras.
   187→            Rendering Instructions: Extremely detailed rendering, volumetric lighting, photorealistic textures, 8k resolution concept art. No watermarks, no embedded text, no logos, no typography. Perfect aspect ratio for eBook cover (2:3 aspect ratio).
   188→        """.trimIndent()
   189→    }

   191→    /**
   192→     * Main EPUB 3 generation pipeline, returning whether export was successful
   193→     */
   194→    fun exportToEpub(
   195→        context: Context, 
   196→        story: Story,
   197→        authorName: String = "",
   198→        authorPenName: String = "",
   199→        authorBio: String = "",
   200→        authorPublisher: String = "",
   201→        authorWebsite: String = "",
   202→        authorSocial: String = "",
   203→        subtitle: String = "",
   204→        seriesTitle: String = "",
   205→        volumeNumber: String = "",
   206→        edition: String = "First Edition",
   207→        isbn: String = "",
   208→        dedication: String = "",
   209→        acknowledgements: String = "",
   210→        audioFiles: List<File>? = null
   211→    ): Boolean {
   212→        val cleanTitle = story.title.replace("[^a-zA-Z0-9]".toRegex(), "_")
   213→        val fileName = "${cleanTitle}_eBook.epub"
   214→        var savedSuccess = false
   215→        
   216→        try {
   217→            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
   218→                val resolver = context.contentResolver
   219→                val contentValues = ContentValues().apply {
   220→                    put(MediaStore.Downloads.DISPLAY_NAME, fileName)
   221→                    put(MediaStore.Downloads.MIME_TYPE, "application/epub+zip")
   222→                    put(MediaStore.Downloads.RELATIVE_PATH, Environment.DIRECTORY_DOWNLOADS)
   223→                }
   224→                val uri = resolver.insert(MediaStore.Downloads.EXTERNAL_CONTENT_URI, contentValues)
   225→                if (uri != null) {
   226→                    resolver.openOutputStream(uri)?.use { out ->
   227→                        writeEpubToStream(
   228→                            context = context,
   229→                            story = story,
   230→                            out = out,
   231→                            authorName = authorName,
   232→                            authorPenName = authorPenName,
   233→                            authorBio = authorBio,
   234→                            authorPublisher = authorPublisher,
   235→                            authorWebsite = authorWebsite,
   236→                            authorSocial = authorSocial,
   237→                            subtitle = subtitle,
   238→                            seriesTitle = seriesTitle,
   239→                            volumeNumber = volumeNumber,
   240→                            edition = edition,
   241→                            isbn = isbn,
   242→                            dedication = dedication,
   243→                            acknowledgements = acknowledgements
   244→                        )
   245→                    }
   246→                    savedSuccess = true
   247→                }
   248→            } else {
   249→                val downloadsDir = Environment.getExternalStoragePublicDirectory(Environment.DIRECTORY_DOWNLOADS)
   250→                if (!downloadsDir.exists()) downloadsDir.mkdirs()
   251→                val destFile = File(downloadsDir, fileName)
   252→                destFile.outputStream().use { out ->
   253→                    writeEpubToStream(
   254→                        context = context,
   255→                        story = story,
   256→                        out = out,
   257→                        authorName = authorName,
   258→                        authorPenName = authorPenName,
   259→                        authorBio = authorBio,
   260→                        authorPublisher = authorPublisher,
   261→                        authorWebsite = authorWebsite,
   262→                        authorSocial = authorSocial,
   263→                        subtitle = subtitle,
   264→                        seriesTitle = seriesTitle,
   265→                        volumeNumber = volumeNumber,
   266→                        edition = edition,
   267→                        isbn = isbn,
   268→                        dedication = dedication,
   269→                        acknowledgements = acknowledgements
   270→                    )
   271→                }
   272→                savedSuccess = true
   273→            }
   274→        } catch (e: Exception) {
   275→            Log.e("EpubManager", "Failed exporting EPUB", e)
   276→            savedSuccess = false
   277→        }
   278→        return savedSuccess
   279→    }

   315→    fun writeEpubToStream(
   316→        context: Context,
   317→        story: Story,
   318→        out: OutputStream,
   319→        authorName: String,
   320→        authorPenName: String,
   321→        authorBio: String,
   322→        authorPublisher: String,
   323→        authorWebsite: String,
   324→        authorSocial: String,
   325→        subtitle: String,
   326→        seriesTitle: String,
   327→        volumeNumber: String,
   328→        edition: String,
   329→        isbn: String,
   330→        dedication: String,
   331→        acknowledgements: String,
   332→        audioFiles: List<File>? = null
   333→    ) {
   334→        val zos = ZipOutputStream(out)
   335→        
   336→        // 1. mimetype (Must be first, completely uncompressed)
   337→        val mimeEntry = ZipEntry("mimetype").apply { method = ZipEntry.STORED }
   338→        val mimeBytes = "application/epub+zip".toByteArray()
   339→        mimeEntry.size = mimeBytes.size.toLong()
   340→        mimeEntry.crc = getCrc(mimeBytes)
   341→        zos.putNextEntry(mimeEntry)
   342→        zos.write(mimeBytes)
   343→        zos.closeEntry()
   344→
   345→        // 2. META-INF/container.xml
   346→        zos.putNextEntry(ZipEntry("META-INF/container.xml"))
   347→        val containerXml = """<?xml version="1.0"?>
   348→<container version="1.0" xmlns="urn:oasis:names:tc:opendocument:xmlns:container">
   349→    <rootfiles>
   350→        <rootfile full-path="OEBPS/content.opf" media-type="application/oebps-package+xml"/>
   351→    </rootfiles>
   352→</container>""".trimIndent()
   353→        zos.write(containerXml.toByteArray())
   354→        zos.closeEntry()
   355→
   356→        // Setup dynamic properties
   357→        val primaryAuthor = authorPenName.ifEmpty { authorName.ifEmpty { "Unknown Author" } }
   358→        val publisherVal = authorPublisher.ifEmpty { "Audio Novel Studio" }
   359→        val finalSubtitle = subtitle.ifEmpty { story.genres + " Novel" }
   360→        val currentYear = SimpleDateFormat("yyyy", Locale.US).format(Date())
   361→        val isoDate = SimpleDateFormat("yyyy-MM-dd'T'HH:mm:ss'Z'", Locale.US).apply {
   362→            timeZone = TimeZone.getTimeZone("UTC")
   363→        }.format(Date())
   364→        val bookUuid = "urn:uuid:" + UUID.nameUUIDFromBytes(story.title.toByteArray()).toString()
   365→        val coverFile = File(context.filesDir, "story_${story.id}_cover.jpg")
   366→        val hasCoverFile = coverFile.exists()
   367→
   368→        // 3. Write CSS Stylesheet
   369→        zos.putNextEntry(ZipEntry("OEBPS/css/stylesheet.css"))
   370→        val cssContent = """
   371→            body {
   372→                font-family: "Georgia", "Times New Roman", serif;
   373→                line-height: 1.6;
   374→                margin: 5% 8%;
   375→                color: #111111;
   376→                background-color: #FFFFFF;
   377→            }
   378→            h1, h2, h3 {
   379→                font-family: "Helvetica Neue", "Arial", sans-serif;
   380→                text-align: center;
   381→                color: #2c3e50;
   382→                margin-top: 1.5em;
   383→                margin-bottom: 0.8em;
   384→            }
   385→            h1 {
   386→                font-size: 1.8em;
   387→                border-bottom: 1px solid #ecf0f1;
   388→                padding-bottom: 0.3em;
   389→            }
   390→            h2 {
   391→                font-size: 1.4em;
   392→            }
   393→            p {
   394→                text-indent: 1.5em;
   395→                margin: 0;
   396→                text-align: justify;
   397→            }
   398→            p.first-paragraph {
   399→                text-indent: 0;
   400→                margin-top: 1.5em;
   401→            }
   402→            p.first-paragraph:first-letter {
   403→                font-size: 3.2em;
   404→                float: left;
   405→                margin: 0.1em 0.1em 0 0;
   406→                line-height: 0.85;
   407→                font-weight: bold;
   408→                color: #2c3e50;
   409→                font-family: "Georgia", serif;
   410→            }
   411→            .scene-break {
   412→                text-align: center;
   413→                margin: 2em 0;
   414→                font-size: 1.4em;
   415→                letter-spacing: 0.5em;
   416→                color: #7f8c8d;
   417→            }
   418→            .center-text {
   419→                text-align: center;
   420→                text-indent: 0;
   421→            }
   422→            .dedication {
   423→                font-style: italic;
   424→                text-align: center;
   425→                margin-top: 5em;
   426→                line-height: 1.8;
   427→            }
   428→            .copyright-info {
   429→                font-size: 0.9em;
   430→                color: #34495e;
   431→                line-height: 1.8;
   432→                text-indent: 0;
   433→                margin-bottom: 1em;
   434→            }
   435→            .about-author-section {
   436→                margin-top: 2em;
   437→            }
   438→            .cover-img {
   439→                display: block;
   440→                max-width: 100%;
   441→                max-height: 100%;
   442→                margin: 0 auto;
   443→            }
   444→        """.trimIndent()
   445→        zos.write(cssContent.toByteArray())
   446→        zos.closeEntry()
   447→
   448→        // 4. Write cover image if it exists
   449→        if (hasCoverFile) {
   450→            zos.putNextEntry(ZipEntry("OEBPS/images/cover.jpg"))
   451→            coverFile.inputStream().use { input ->
   452→                input.copyTo(zos)
   453→            }
   454→            zos.closeEntry()
   455→        }
   456→
   457→        // 5. OEBPS/content.opf (Manifest and Spine)
   458→        zos.putNextEntry(ZipEntry("OEBPS/content.opf"))
   459→        val chapters = story.getChapters()
   460→        val manifestItems = StringBuilder()
   461→        val spineItems = StringBuilder()
   462→
   463→        chapters.forEachIndexed { idx, _ ->
   464→            val hasAudio = audioFiles != null && audioFiles.size > idx
   465→            val mediaOverlay = if (hasAudio) " media-overlay=\"smil-chapter${idx+1}\"" else ""
   466→            
   467→            manifestItems.append("""<item id="chapter${idx+1}" href="chapter${idx+1}.xhtml" media-type="application/xhtml+xml"$mediaOverlay/>""").append("\n        ")
   468→            
   469→            if (hasAudio) {
   470→                val audioFile = audioFiles!![idx]
   471→                val ext = audioFile.extension.ifEmpty { "mp3" }
   472→                val mediaType = if (ext.lowercase() == "wav") "audio/wav" else "audio/mpeg"
   473→                manifestItems.append("""<item id="smil-chapter${idx+1}" href="smil/chapter${idx+1}.smil" media-type="application/smil+xml"/>""").append("\n        ")
   474→                manifestItems.append("""<item id="audio-chapter${idx+1}" href="audio/chapter${idx+1}.$ext" media-type="$mediaType"/>""").append("\n        ")
   475→            }
   476→            
   477→            spineItems.append("""<itemref idref="chapter${idx+1}"/>""").append("\n        ")
   478→        }
   479→
   480→        val coverImageManifestItem = if (hasCoverFile) {
   481→            """<item id="cover-image" href="images/cover.jpg" media-type="image/jpeg" properties="cover-image"/>"""
   482→        } else ""
   483→
   484→        val contentOpf = """<?xml version="1.0" encoding="UTF-8"?>
   485→<package xmlns="http://www.idpf.org/2007/opf" unique-identifier="BookId" version="3.0">
   486→    <metadata xmlns:dc="http://purl.org/dc/elements/1.1/">
   487→        <dc:title>${story.title}</dc:title>
   488→        <dc:creator id="creator">${primaryAuthor}</dc:creator>
   489→        <dc:language>en</dc:language>
   490→        <dc:identifier id="BookId">${bookUuid}</dc:identifier>
   491→        <dc:publisher>${publisherVal}</dc:publisher>
   492→        <dc:date>${isoDate}</dc:date>
   493→        <dc:description>${story.metaDesc.ifEmpty { story.hook }}</dc:description>
   494→        <dc:subject>${story.genres}</dc:subject>
   495→        <meta property="dcterms:modified">${isoDate}</meta>
   496→        ${if (hasCoverFile) """<meta name="cover" content="cover-image"/>""" else ""}
   497→    </metadata>
   498→    <manifest>
   499→        <item id="stylesheet" href="css/stylesheet.css" media-type="text/css"/>
   500→        <item id="nav" href="nav.xhtml" media-type="application/xhtml+xml" properties="nav"/>
   501→        <item id="ncx" href="toc.ncx" media-type="application/x-dtbncx+xml"/>
   502→        <item id="cover-page" href="cover.xhtml" media-type="application/xhtml+xml"/>
   503→        <item id="title-page" href="title_page.xhtml" media-type="application/xhtml+xml"/>
   504→        <item id="copyright-page" href="copyright.xhtml" media-type="application/xhtml+xml"/>
   505→        <item id="dedication-page" href="dedication.xhtml" media-type="application/xhtml+xml"/>
   506→        $manifestItems
   507→        <item id="acknowledgements-page" href="acknowledgements.xhtml" media-type="application/xhtml+xml"/>
   508→        <item id="about-author-page" href="about_author.xhtml" media-type="application/xhtml+xml"/>
   509→        <item id="promo-page" href="promo.xhtml" media-type="application/xhtml+xml"/>
   510→        $coverImageManifestItem
   511→    </manifest>
   512→    <spine toc="ncx">
   513→        <itemref idref="cover-page"/>
   514→        <itemref idref="title-page"/>
   515→        <itemref idref="copyright-page"/>
   516→        <itemref idref="dedication-page"/>
   517→        $spineItems
   518→        <itemref idref="acknowledgements-page"/>
   519→        <itemref idref="about-author-page"/>
   520→        <itemref idref="promo-page"/>
   521→    </spine>
   522→</package>""".trimIndent()
   523→        zos.write(contentOpf.toByteArray())
   524→        zos.closeEntry()
   525→
   526→        // 6. OEBPS/nav.xhtml (MANDATORY EPUB 3 Navigation Document)
   527→        zos.putNextEntry(ZipEntry("OEBPS/nav.xhtml"))
   528→        val navChapters = StringBuilder()
   529→        chapters.forEachIndexed { idx, _ ->
   530→            navChapters.append("""<li><a href="chapter${idx+1}.xhtml">Chapter ${idx+1}</a></li>""").append("\n            ")
   531→        }
   532→
   533→        val navXhtml = """<?xml version="1.0" encoding="UTF-8"?>
   534→<html xmlns="http://www.w3.org/1999/xhtml" xmlns:epub="http://www.idpf.org/2007/ops" lang="en" xml:lang="en">
   535→<head>
   536→    <title>Navigation</title>
   537→    <meta charset="utf-8" />
   538→    <link rel="stylesheet" href="css/stylesheet.css" type="text/css"/>
   539→</head>
   540→<body>
   541→    <nav epub:type="toc" id="toc">
   542→        <h1>Table of Contents</h1>

   866→    private fun getCrc(bytes: ByteArray): Long {
   867→        val crc32 = java.util.zip.CRC32()
   868→        crc32.update(bytes)
